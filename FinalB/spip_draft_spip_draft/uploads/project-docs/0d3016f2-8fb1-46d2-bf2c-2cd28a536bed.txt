function getStatusMeta(status) {
  const normalized = (status || 'PENDING').toUpperCase();

  if (normalized === 'APPROVED') {
    return {
      label: 'Approved',
      className: 'badge badge-success',
      accent: '#47c59a',
      icon: FiCheckCircle,
    };
  }

  if (normalized === 'REJECTED') {
    return {
      label: 'Rejected',
      className: 'badge badge-rose',
      accent: '#d67a8e',
      icon: FiAlertCircle,
    };
  }

  return {
    label: normalized === 'UNDER_REVIEW' ? 'Under Review' : 'Pending',
    className: 'badge badge-warning',
    accent: '#f2b563',
    icon: FiClock,
  };
}

function getProgress(project) {
  const total = Number(project?.totalPlots || 0);
  const available = Number(project?.availablePlots || 0);

  if (total <= 0) return 0;
  const sold = Math.max(total - available, 0);
  return Math.min((sold / total) * 100, 100);
}

    form.projectName,
    form.description,
    form.location,
    form.city,
    form.state,
    form.totalPlots,
    form.availablePlots,
    form.reraRegistrationNumber,
  ].filter((v) => String(v || '').trim() !== '').length;

  const totalMainFields = 8;

  const hasMeaningfulReraStatus =
    form.reraStatus && form.reraStatus !== 'NOT_SUBMITTED';

  const hasMeaningfulDcStatus =
    form.dcConversionStatus && form.dcConversionStatus !== 'NOT_APPLICABLE';

  const uploadedDocsCount = (documents || []).filter((doc) => !!doc.file).length;

  let score = filledMainFields;
  if (hasMeaningfulReraStatus) score += 1;
  if (hasMeaningfulDcStatus) score += 1;

  const totalPossible = totalMainFields + 1 + 1 + 3;

function AmenitiesPicker({ amenities, setAmenities, ui, theme }) {
  const toggleAmenity = (value) => {
    setAmenities((prev) =>
      prev.includes(value)
        ? prev.filter((item) => item !== value)
        : [...prev, value]
    );
  };

  return (
    <div>
      <div className="flex flex-wrap gap-3">
        {PROJECT_AMENITIES.map((item) => {
          const active = amenities.includes(item.value);

          return (
            <button
              key={item.value}
              type="button"
              onClick={() => toggleAmenity(item.value)}
              className="px-4 py-2.5 rounded-2xl text-sm font-bold transition-all"
              style={{
                background: active
                  ? 'linear-gradient(135deg, #174b72 0%, #267c8e 55%, #3B9797 100%)'
                  : ui.cardBg,
                color: active ? '#ffffff' : ui.textPrimary,
                border: active ? '1px solid rgba(59,151,151,0.34)' : ui.border,
                boxShadow: active
                  ? '0 10px 24px rgba(22,71,106,0.18)'
                  : 'none',
                cursor: 'pointer',
              }}
            >
              {item.label}
            </button>
          );
        })}
      </div>

      <div className="mt-4">
        <p
          className="text-[11px] uppercase tracking-[0.12em] font-bold"
          style={{ color: ui.textMuted }}
        >
          Selected Amenities
        </p>
        <div className="flex flex-wrap gap-2 mt-3">
          {amenities.length > 0 ? (
            amenities.map((item) => (
              <span
                key={item}
                className="px-3 py-2 rounded-xl text-xs font-bold"
                style={{
                  background:
                    theme === 'dark' ? 'rgba(59,151,151,0.12)' : '#e8f7f5',
                  color: ui.textPrimary,
                  border: '1px solid rgba(59,151,151,0.22)',
                }}
              >
                {String(item).replaceAll('_', ' ')}
              </span>
            ))
          ) : (
            <span className="text-sm" style={{ color: ui.textMuted }}>
              No amenities selected yet.
            </span>
          )}
        </div>
      </div>
    </div>
  );
}

function CreateProjectModal({ onClose, onCreated, ui, theme }) {
  const [form, setForm] = useState({
    projectName: '',
    description: '',
    location: '',
    city: '',
    state: '',
    totalPlots: '',
    availablePlots: '',
    reraRegistrationNumber: '',
    reraStatus: 'NOT_SUBMITTED',
    dcConversionStatus: 'NOT_APPLICABLE',
    legalNotes: '',
  });

  const [amenities, setAmenities] = useState([]);
  const [documents, setDocuments] = useState(
    PROJECT_DOCUMENT_TYPES.map((item) => ({
      type: item.type,
      label: item.label,
      file: null,
    }))
  );

  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const [successMsg, setSuccessMsg] = useState('');
  const [projectImage, setProjectImage] = useState(null);

  const set = (k, v) => setForm((f) => ({ ...f, [k]: v }));

  const validate = () => {
    if (!form.projectName.trim()) return 'Project name is required.';
    if (!form.description.trim()) return 'Description is required.';
    if (!form.location.trim()) return 'Location is required.';
    if (!form.city.trim()) return 'City is required.';
    if (!form.state.trim()) return 'State is required.';
    if (!form.totalPlots || +form.totalPlots < 1) {
      return 'Total plots must be at least 1.';
    }
    if (form.availablePlots === '' || +form.availablePlots < 0) {
      return 'Available plots cannot be negative.';
    }
    if (+form.availablePlots > +form.totalPlots) {
      return 'Available plots cannot exceed total plots.';
    }
    return '';
  };

  const handleSubmit = async () => {
    const err = validate();
    if (err) {
      setError(err);
      return;
    }

    setSaving(true);
    setError('');
    setSuccessMsg('');

    try {
      const projectRes = await createProject({
        projectName: form.projectName,
        description: form.description,
        location: form.location,
        city: form.city,
        state: form.state,
        totalPlots: +form.totalPlots,
        availablePlots: +form.availablePlots,
      });

      const createdProject = projectRes?.data?.project || projectRes?.data;
      const backendMessage =
        projectRes?.data?.message || 'Project created successfully.';
      const kycStatus = projectRes?.data?.kycStatus || null;

      if (!createdProject?.id) {
        throw new Error('Project created but project ID was not returned.');
      }

      await saveProjectLegalDetails(createdProject.id, {
        reraRegistrationNumber: form.reraRegistrationNumber || null,
        reraStatus: form.reraStatus,
        dcConversionStatus: form.dcConversionStatus,
        legalNotes: form.legalNotes || null,
      });

      if (amenities.length > 0) {
        await addAmenitiesBulk(
          createdProject.id,
          amenities.map((item) => ({
            amenityName: item,
          }))
        );
      }

      for (const doc of documents) {
        if (doc.file) {
          await uploadProjectDocument(createdProject.id, doc.type, doc.file);
        }
      }

      setSuccessMsg(backendMessage);
      onCreated(createdProject, backendMessage, kycStatus);

      setTimeout(() => {
        onClose();
      }, 900);
    } catch (e) {
      setError(
        e?.response?.data?.message ||
          e?.response?.data?.error ||
          e?.message ||
          'Failed to create project. Please try again.'
      );
    } finally {
      setSaving(false);
    }
  };

  const stats = [
    {
      label: 'Project Status',
      value: 'Draft Setup',
      icon: FiBriefcase,
    },
    {
      label: 'Compliance Track',
      value:
        form.reraStatus === 'APPROVED'
          ? 'Strong'
          : form.reraStatus === 'PENDING'
          ? 'In Progress'
          : 'Pending',
      icon: FiShield,
    },
    {
      label: 'Inventory Scope',
      value: form.totalPlots ? `${form.totalPlots} plots` : 'Not set',
      icon: FiLayers,
    },
    {
      label: 'Amenities',
      value: amenities.length > 0 ? `${amenities.length} selected` : 'Not set',
      icon: FiGrid,
    },
  ];

  return (
    <Modal
      title="Create New Project"
      subtitle="Enter the project details, provide legal information, select amenities, and upload required documents for approval."
      onClose={onClose}
      ui={ui}
      theme={theme}
    >
      <div className="space-y-6">
        <div
          className="rounded-[28px] p-4 md:p-5"
          style={{
            background: ui.cardBg,
            border: ui.border,
          }}
        >
          <div className="grid grid-cols-1 lg:grid-cols-[1.1fr_0.9fr] gap-4 items-start">
            <div>
              <p
                className="text-[11px] uppercase font-extrabold tracking-[0.14em]"
                style={{
                  color:
                    theme === 'dark'
                      ? 'rgba(159,221,228,0.86)'
                      : '#2b7a8b',
                }}
              >
                Creation Guide
              </p>
              <h4
                className="font-extrabold text-lg mt-1"
                style={{ color: ui.textPrimary }}
              >
                Project details first. Then legal, amenities, and documents.
              </h4>
              <p
                className="text-sm leading-6 mt-2"
                style={{ color: ui.textSecondary }}
              >
                Complete each section carefully. Your project will be reviewed by admin before it is published.
              </p>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              {stats.map(({ label, value, icon: Icon }) => (

                  key={label}

function BuilderProjects() {
  const { theme } = useTheme();
  const ui = getProjectThemeStyles(theme);
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();

  const [projects, setProjects] = useState([]);
  const [loading, setLoading] = useState(false);
  const [search, setSearch] = useState('');
  const [filter, setFilter] = useState('ALL');
  const [showCreateModal, setShowCreateModal] = useState(false);
  const [showLegalModal, setShowLegalModal] = useState(false);
  const [selectedProject, setSelectedProject] = useState(null);

  useEffect(() => {
    loadProjects();
  }, []);

  useEffect(() => {
    if (searchParams.get('action') === 'create') {
      setShowCreateModal(true);
    }
  }, [searchParams]);

  const loadProjects = async () => {
    try {
      setLoading(true);
      const res = await getMyProjects();
      const projectData = Array.isArray(res?.data)
        ? res.data
        : res?.data?.projects || [];
      setProjects(projectData);
    } catch (error) {
      console.error('Failed to load projects:', error);
      setProjects([]);
    } finally {
      setLoading(false);
    }
  };

  const filteredProjects = useMemo(() => {
    return projects.filter((project) => {
      const status = (project?.approvalStatus || project?.status || '').toUpperCase();

      const matchesFilter =
        filter === 'ALL'
          ? true
          : filter === 'PENDING'
          ? status === 'PENDING' || status === 'UNDER_REVIEW'
          : status === filter;

      const q = search.trim().toLowerCase();
      const matchesSearch =
        !q ||
        (project?.projectName || '').toLowerCase().includes(q) ||
        (project?.location || '').toLowerCase().includes(q) ||
        (project?.city || '').toLowerCase().includes(q) ||
        (project?.state || '').toLowerCase().includes(q);

      return matchesFilter && matchesSearch;
    });
  }, [projects, search, filter]);

  const approvedCount = projects.filter(
    (p) => (p?.approvalStatus || p?.status || '').toUpperCase() === 'APPROVED'
  ).length;

  const pendingCount = projects.filter((p) => {
    const s = (p?.approvalStatus || p?.status || '').toUpperCase();
    return s === 'PENDING' || s === 'UNDER_REVIEW';
  }).length;

  return (
    <div
      className="min-h-screen"
      style={{
        background:
          theme === 'dark'
            ? 'linear-gradient(180deg, #081120 0%, #0a1730 100%)'
            : 'linear-gradient(180deg, #f4f8fc 0%, #eef4f9 100%)',
        color: ui.textPrimary,
      }}
    >
      <BuilderSidebar />

      <main className="lg:ml-[280px] px-4 md:px-6 xl:px-8 py-6 md:py-8">
        <PremiumPageHero
          onOpenCreate={() => setShowCreateModal(true)}
          totalProjects={projects.length}
          approvedCount={approvedCount}
          pendingCount={pendingCount}
          ui={ui}
          theme={theme}
        />

        <PremiumFilterBar
          search={search}
          setSearch={setSearch}
          filter={filter}
          setFilter={setFilter}
          ui={ui}
          theme={theme}
        />

        {loading ? (
          <div
            className="rounded-[24px] p-8"
            style={{
              background: ui.cardBg,
              border: ui.border,
              color: ui.textSecondary,
            }}
          >
            Loading projects...
          </div>
        ) : filteredProjects.length === 0 ? (
          <EmptyState onOpenCreate={() => setShowCreateModal(true)} ui={ui} theme={theme} />
        ) : (
          <div className="grid grid-cols-1 xl:grid-cols-2 2xl:grid-cols-3 gap-5">
            {filteredProjects.map((project) => (
              <ProjectCard
                key={project.id}
                project={project}
                onOpenLegal={(selectedProject) => {
                  setSelectedProject(selectedProject);
                  setShowLegalModal(true);
                }}
                ui={ui}
                theme={theme}
              />
            ))}
          </div>
        )}

        {showCreateModal ? (
          <CreateProjectModal
            onClose={() => setShowCreateModal(false)}
            onCreated={(createdProject) => {
              setProjects((prev) => [createdProject, ...prev]);
              setShowCreateModal(false);
            }}
            ui={ui}
            theme={theme}
          />
        ) : null}

        {showLegalModal && selectedProject ? (
          <ProjectLegalModal
            project={selectedProject}
            onClose={() => {
              setSelectedProject(null);
              setShowLegalModal(false);
            }}
            onSaved={(updatedProject) => {
              setProjects((prev) =>
                prev.map((item) =>
                  item.id === updatedProject.id ? updatedProject : item
                )
              );
              setSelectedProject(updatedProject);
            }}
          />
        ) : null}
      </main>
    </div>
  );
}

export default BuilderProjects;
